! 北京 (bj)  站点 ID = 1

 address-family ipv6
  route-distinguisher 9425:1
  vpn-target 9425:1
  apply-label per-instance

interface 100GE3/0/40.3501
 vlan-type dot1q 3501
 ip binding vpn-instance zslice0001
 ipv6 address 240a:a806:1:101::1/64

ipv6 route-static vpn-instance zslice0001 240a:a806:1:1:: 64 240a:a806:1:101::2 tag 1

ip vpn-instance zslice0002
 address-family ipv6
  route-distinguisher 9425:2
  vpn-target 9425:2
  apply-label per-instance

interface 100GE3/0/40.3502
 vlan-type dot1q 3502
 ip binding vpn-instance zslice0002
 ipv6 address 240a:a806:2:101::1/64

ipv6 route-static vpn-instance zslice0002 240a:a806:2:1:: 64 240a:a806:2:101::2 tag 2

vsi zslice0003
 evpn encapsulation srv6
  route-distinguisher 9425:3
  vpn-target 9425:3
  segment-routing ipv6 best-effort
  segment-routing ipv6 locator bj-e1

interface ten-gigabitethernet 3/1/1.3503
 vlan-type dot1q vid 3503
 xconnect vsi zslice0003

vsi zslice0004
 evpn encapsulation srv6
  route-distinguisher 9425:4
  vpn-target 9425:4
  segment-routing ipv6 best-effort
  segment-routing ipv6 locator bj-e1

interface ten-gigabitethernet 3/1/1.3504
 vlan-type dot1q vid 3504
 xconnect vsi zslice0004
